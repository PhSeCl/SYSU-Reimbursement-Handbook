% Lower each 20pt widget so its centered text shares the label's baseline.
% Optional alignment uses the standard PDF values: 0 = left, 1 = center.
\newcommand{\Entry}[5][0]{%
  \ifdefined\FormExample
    \ifnum#1=1
      \makebox[#4][c]{\csname Value#2\endcsname}%
    \else
      \makebox[#4][l]{\csname Value#2\endcsname}%
    \fi
  \else
    \raisebox{-5pt}{\TextField[name=#2,altname={#3},width=#4,height=#5,align=#1,
      charsize=12pt,borderwidth=0.5,bordercolor={0 0 0},
      backgroundcolor={1 1 1},color={0 0 0}]{}%
    }%
  \fi
}
\newcommand{\LongEntry}[4]{%
  \ifdefined\FormExample
    \parbox[t][#4][t]{#3}{\raggedright\csname Value#1\endcsname}%
  \else
    \TextField[name=#1,altname={#2},width=#3,height=#4,multiline=true,
      charsize=12pt,borderwidth=0.5,bordercolor={0 0 0},
      backgroundcolor={1 1 1},color={0 0 0}]{}%
  \fi
}
\newcommand{\Block}[3]{%
  #2\par\vspace{2mm}%
  \LongEntry{#1}{#2}{\dimexpr\linewidth-2pt\relax}{#3}\par
}
